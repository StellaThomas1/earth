// NASA (public domain) world topo/bathy texture (Mercator) const MAP_URL = "https://eoimages.gsfc.nasa.gov/images/imagerecords/73000/73909/world.topo.bathy.200412.3x5400x2700.jpg"; const canvas = document.getElementById("map"); const ctx = canvas.getContext("2d", { alpha: true }); const elSystems = document.getElementById("systems"); const elDoctrine = document.getElementById("doctrine"); const elWhisper = document.getElementById("whisper"); let W=0, H=0, dpr=1; function resize(){ dpr = Math.min(window.devicePixelRatio || 1, 2); W = Math.floor(window.innerWidth); H = Math.floor(window.innerHeight); canvas.width = Math.floor(W * dpr); canvas.height = Math.floor(H * dpr); canvas.style.width = W + "px"; canvas.style.height = H + "px"; ctx.setTransform(dpr,0,0,dpr,0,0); } window.addEventListener("resize", resize, { passive:true }); resize(); const img = new Image(); img.crossOrigin = "anonymous"; img.src = MAP_URL; // cinematic fade-in + subtle lift let t = 0; function ease(x){ return x*x*(3-2*x); } function draw(){ ctx.clearRect(0,0,W,H); // background wash (keeps blacks rich even before image loads) const g = ctx.createRadialGradient(W*0.55, H*0.40, 0, W*0.55, H*0.40, Math.max(W,H)*0.9); g.addColorStop(0, "rgba(12,16,28,0.55)"); g.addColorStop(1, "rgba(0,0,0,0)"); ctx.fillStyle = g; ctx.fillRect(0,0,W,H); if (img.complete && img.naturalWidth){ // progress t = Math.min(1, t + 0.008); const s = ease(t); // fit image to canvas (cover) const iw = img.naturalWidth, ih = img.naturalHeight; const scale = Math.max(W/iw, H/ih); const dw = iw * scale; const dh = ih * scale; // gentle lift as it fades in (coming online) const lift = (1 - s) * 18; // px const dx = (W - dw)/2; const dy = (H - dh)/2 + lift; // darken + color grade (quiet Davos) ctx.globalAlpha = 0.85 * s; ctx.filter = `contrast(${1.05}) saturate(${0.95 + 0.10*s}) brightness(${0.55 + 0.35*s})`; ctx.drawImage(img, dx, dy, dw, dh); // subtle coastline glow pass ctx.globalAlpha = 0.12 * s; ctx.filter = "none"; ctx.fillStyle = "rgba(170,240,255,0.10)"; ctx.fillRect(0,0,W,H); // reset ctx.globalAlpha = 1; ctx.filter = "none"; // reveal typography on schedule if (t > 0.22) elSystems.classList.add("on"); if (t > 0.36) elDoctrine.classList.add("on"); if (t > 0.55) elWhisper.classList.add("on"); } requestAnimationFrame(draw); } img.onload = () => { // start render loop when image is ready draw(); }; img.onerror = () => { // If the NASA image blocks (rare), still show typography elSystems.classList.add("on"); elDoctrine.classList.add("on"); elWhisper.classList.add("on"); };
